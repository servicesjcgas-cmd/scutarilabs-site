// Add to pubspec.yaml: scutari_audio_core: ^1.0.0
import 'package:scutari_audio_core/scutari_audio_core.dart';

void main() async {
  // Initialize strictly on-device engine with offline Ed25519 license key
  final engine = await ScutariAudioCore.init(
    licenseKey: 'SCUTARI_ED25519_KEY_...',
    mode: InferenceMode.strictOffline,
    delegate: HardwareDelegate.armNpu,
  );

  // Stream real-time classified threats without transmitting raw audio
  engine.threatStream.listen((event) {
    print('Local Acoustic Event: ${event.classification} [${event.confidence}]');
    HapticsService.triggerStealthAlert(event.severity);
  });
}
